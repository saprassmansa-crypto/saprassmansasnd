//================= API =================
async function api(action, extra){
  if(GAS_URL.includes("PASTE_URL_WEBAPP")){ toast("GAS_URL belum diisi di index.html", true); throw new Error("config"); }
  const body = Object.assign({ action: action, currentUserId: S?.user?.id, currentUserRole: S?.user?.role, currentUserName: S?.user?.nama }, extra||{});
  let res, json;
  try {
    res = await fetch(GAS_URL, { method:"POST", headers:{"Content-Type":"text/plain;charset=utf-8"}, body: JSON.stringify(body) });
    json = await res.json();
  } catch(err){
    toast("Gagal menghubungi server. Periksa URL Apps Script / koneksi internet.", true);
    throw err;
  }
  if(json.status === "error"){ toast(json.message || "Terjadi kesalahan server.", true); throw new Error(json.message); }
  return json;
}

//================= LOGIN / LOGOUT =================
async function doLogin(){
  const username = $("lg-user").value.trim(), password = $("lg-pass").value.trim();
  if(!username || !password){ $("lg-msg").textContent = "Isi username dan password."; return; }
  $("lg-msg").textContent = "";
  loading(true, "Memeriksa akun...");
  try {
    const j = await api("login", { username, password });
    S = { user: j.data };
    sessionStorage.setItem("sapras_session", JSON.stringify(S));
    await boot();
  } catch(e){ $("lg-msg").textContent = e.message || "Gagal masuk."; }
  loading(false);
}
function logout(){ S = null; sessionStorage.removeItem("sapras_session"); $("app").classList.add("hide"); $("login").classList.remove("hide"); }

async function saveAset(id){
  const kat = D.categories.find(x=>x.ID===$("f-kat").value), lok = D.locations.find(x=>x.ID===$("f-lok").value);
  if(!$("f-nama").value.trim() || !kat || !lok){ toast("Nama, kategori, dan lokasi wajib diisi.", true); return; }
  const asset = { KodeBarang: $("f-kode").value.trim(), NamaBarang: $("f-nama").value.trim(), KategoriID: kat.ID, Kategori: kat.NamaKategori, LokasiID: lok.ID, Lokasi: lok.NamaLokasi, Jumlah: $("f-jml").value, Satuan: $("f-sat").value, Kondisi: $("f-kon").value, TahunPerolehan: $("f-thn").value, SumberDana: $("f-dana").value, HargaPerolehan: $("f-harga").value, Keterangan: $("f-ket").value.trim() };
  if(id) asset.ID = id;
  loading(true, "Menyimpan...");
  try {
    const file = $("f-foto").files[0];
    if(file){
      const base64 = await fileToBase64(file);
      const up = await api("uploadFile", { fileName: file.name, base64Data: base64, folderType: "Foto Aset" });
      asset.FotoFileID = up.data.fileId; asset.FotoURL = up.data.fileUrl;
    }
    await api(id ? "updateAsset" : "saveAsset", { asset });
    await refreshAll();
    closeModal(); toast("Aset berhasil disimpan."); switchTab("aset");
  } catch(e){} finally { loading(false); }
}
function fileToBase64(f){ return new Promise((res,rej)=>{ const r=new FileReader(); r.onload=()=>res(r.result); r.onerror=rej; r.readAsDataURL(f); }); }
async function delAset(id){
  const c = await Swal.fire({ title:"Hapus aset ini?", icon:"warning", showCancelButton:true, confirmButtonText:"Ya, Hapus", confirmButtonColor:"#dc2626" });
  if(!c.isConfirmed) return;
  loading(true);
  try { await api("deleteAsset", { id }); await refreshAll(); toast("Aset dihapus."); switchTab("aset"); } catch(e){} finally { loading(false); }
}
function printBarcode(kode, nama){
  if(!kode){ toast("Aset ini belum punya kode barang.", true); return; }
  const w = window.open("", "_blank", "width=420,height=340");
  if(!w){ toast("Popup diblokir browser.", true); return; }
  const html = "<!DOCTYPE html><html><head><title>Barcode</title>" +
    "<script src='https://cdn.jsdelivr.net/npm/jsbarcode@3.11.6/dist/JsBarcode.all.min.js'><\/script>" +
    "<style>body{font-family:sans-serif;text-align:center;padding:24px}button{margin-top:16px;padding:10px 20px;border:0;border-radius:6px;background:#1e3a8a;color:#fff;font-weight:bold}@media print{button{display:none}}</style></head><body>" +
    "<div style='font-weight:bold;margin-bottom:8px'>" + esc(nama||"") + "</div><svg id='bc'></svg>" +
    "<div><button onclick='window.print()'>CETAK</button></div>" +
    "<script>JsBarcode('#bc'," + JSON.stringify(String(kode)) + ",{format:'CODE128',width:2,height:60,fontSize:14,margin:8});<\/script></body></html>";
  w.document.write(html); w.document.close();
}

async function simpanSettings(){
  const keys = ["NamaSekolah","AlamatSekolah","Desa","Kecamatan","Kabupaten","Provinsi","NPSN","Email","Telepon","NamaKepalaSekolah","NIPKepalaSekolah","NamaPengurusBarang","NIPPengurusBarang"];
  const settings = {}; keys.forEach(k => settings[k] = $("s-"+k).value.trim());
  loading(true, "Menyimpan pengaturan...");
  try {
    const fs = $("lg-sekolah").files[0], fd = $("lg-daerah").files[0];
    if(fs){ const b = await fileToBase64(fs); const up = await api("uploadFile", { fileName:fs.name, base64Data:b, folderType:"Logo" }); settings.LogoSekolahURL = up.data.fileUrl; }
    if(fd){ const b = await fileToBase64(fd); const up = await api("uploadFile", { fileName:fd.name, base64Data:b, folderType:"Logo" }); settings.LogoDaerahURL = up.data.fileUrl; }
    await api("saveSettings", { settings });
    await refreshAll(); toast("Pengaturan disimpan."); switchTab("pengaturan");
  } catch(e){} finally { loading(false); }
}
